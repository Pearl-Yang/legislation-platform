QwenFacade.java
// Qwen 在线/离线双通道
public class QwenFacade {
  public String execute(String prompt,
    Supplier<String> fallback) {
    try {
      // 在线模式
      return callQwenAPI(prompt);
    } catch (Exception e) {
      // 失败 → 自动降级离线
      log("Qwen 调用失败，降级到 offline");
      return fallback.get();
    }
  }
}

// 业务调用
String content = qwenFacade.execute(
  userPrompt,
  () -> buildPlaceholderContent(
    "XX 智慧城市管理条例",
    Arrays.asList("上位法", "总则")
  )
);
